57std::string shl(const std::string& e, int from, int sh, int to) {
58 int pad = to - from - sh;
59 if (pad < 0) throw std::logic_error("shl: result does not fit");
60 if (sh == 0 && pad == 0) return e;
61 std::string s = e;
62 if (sh > 0) s = s + ", " + zeros(sh);
63 if (pad > 0) s = zeros(pad) + ", " + s;
64 return "{" + s + "}";
65}

120void emit_step(std::ostringstream& os, int w, bool pipe) {
121 Split s = split_of(w);
122 int pw = 2 * w;
123
124 std::string a_low = "a" + rng(s.lo);
125 std::string b_low = "b" + rng(s.lo);
126 std::string a_high = "a[" + num(w - 1) + ":" + num(s.lo) + "]";
127 std::string b_high = "b[" + num(w - 1) + ":" + num(s.lo) + "]";
128 std::string a_sum = zext(a_low, s.lo, s.mid) + " + " + zext(a_high, s.hi, s.mid);
129 std::string b_sum = zext(b_low, s.lo, s.mid) + " + " + zext(b_high, s.hi, s.mid);
130 std::vector<Sig> halves = {{"a0", s.lo, a_low}, {"a1", s.hi, a_high}, {"b0", s.lo, b_low},
131 {"b1", s.hi, b_high}, {"sa", s.mid, a_sum}, {"sb", s.mid, b_sum}};
132 emit_phase(os, pipe, 1, "split, pre-additions");
133 if (pipe) {
134 emit_stage(os, halves);
135 } else {
136 emit_wires(os, halves);
137 }
138
139 std::string out = "z";
140 if (pipe) out = "m";
141 emit_phase(os, pipe, 2, "three half width products");
142 os << " wire " << rng(2 * s.lo) << " " << out << "0;\n";
143 os << " wire " << rng(2 * s.hi) << " " << out << "2;\n";
144 os << " wire " << rng(2 * s.mid) << " " << out << "m;\n";
145 os << " " << mul_module(s.lo) << " u_z0 (.a(a0), .b(b0), .p(" << out << "0));\n";
146 os << " " << mul_module(s.hi) << " u_z2 (.a(a1), .b(b1), .p(" << out << "2));\n";
147 os << " " << mul_module(s.mid) << " u_zm (.a(sa), .b(sb), .p(" << out << "m));\n\n";
148 if (pipe) {
149 emit_stage(os, {{"z0", 2 * s.lo, "m0"}, {"z2", 2 * s.hi, "m2"}, {"zm", 2 * s.mid, "mm"}});
150 }
151
152 emit_phase(os, pipe, 3, "z1 = zm - z2 - z0, recombination");
153 os << " wire " << rng(2 * s.mid) << " zd = zm - " << zext("z2", 2 * s.hi, 2 * s.mid) << " - "
154 << zext("z0", 2 * s.lo, 2 * s.mid) << ";\n";
155 os << " wire " << rng(w + 1) << " z1 = zd" << rng(w + 1) << ";\n";
156 std::string product = shl("z2", 2 * s.hi, 2 * s.lo, pw) + "\n + " +
157 shl("z1", w + 1, s.lo, pw) + "\n + " +
158 zext("z0", 2 * s.lo, pw);
159 if (pipe) {
160 emit_stage(os, {{"pr", pw, product}});
161 os << " assign p = pr;\n";
162 } else {
163 os << " assign p = " << product << ";\n";
164 }
165}

188void emit_leaf(std::ostringstream& os, int w) {
189 std::string comment = "// " + num(w) + " x " + num(w) +
190 " combinational leaf: long multiplication,\n"
191 "// the sum of the partial products ({" +
192 num(w) + "{b[i]}} & a) << i\n";
193 emit_header(os, mul_module(w), w, comment, "");
194 os << " assign p = ";
195 for (int i = 0; i < w; ++i) {
196 if (i > 0) os << "\n + ";
197 std::string partial = "({" + num(w) + "{b[" + num(i) + "]}} & a)";
198 os << shl(partial, w, i, 2 * w);
199 }
200 os << ";\n";
201 os << "endmodule\n\n";
202}

217void emit_top(std::ostringstream& os, int n) {
218 std::string comment =
219 "// " + std::string(kTop) + ": pipelined unsigned multiplier " + num(n) + " x " + num(n) +
220 " -> " + num(2 * n) + "\n" +
221 "// a, b are sampled on a rising edge of clk, their product appears on p\n"
222 "// after the 3rd rising edge. New operands may be applied every clock.\n";
223 emit_header(os, kTop, n, comment, " input wire clk,\n");
224
225 if (n >= 2) {
226 emit_step(os, n, true);
227 } else {
228 emit_stage(os, {{"s1", 1, "a & b"}});
229 emit_stage(os, {{"s2", 1, "s1"}});
230 emit_stage(os, {{"s3", 1, "s2"}});
231 os << " assign p = {1'b0, s3};\n";
232 }
233 os << "endmodule\n";
234}

244Split split_of(int w) {
245 if (w < 2) throw std::invalid_argument("split_of: width must be at least 2");
246 Split s;
247 s.lo = (w + 1) / 2;
248 s.hi = w - s.lo;
249 s.mid = s.lo + 1;
250 return s;
251}

258std::vector<int> required_widths(int n) {
259 validate(n);
260 std::set<int> need;
261 std::vector<int> todo;
262 if (n >= 2) todo.push_back(n);
263 while (!todo.empty()) {
264 int w = todo.back();
265 todo.pop_back();
266 Split s = split_of(w);
267 std::vector<int> parts = {s.lo, s.hi, s.mid};
268 for (int part : parts) {
269 bool is_new = need.count(part) == 0;
270 need.insert(part);
271 if (is_new && part > kLeafWidth) todo.push_back(part);
272 }
273 }
274 return std::vector<int>(need.begin(), need.end());
275}

277std::string generate_rtl(int n) {
278 validate(n);
279 std::ostringstream os;
280 os << "// Unsigned " << n << " x " << n << " -> " << 2 * n
281 << " multiplier, Karatsuba algorithm, 3 stage pipeline.\n";
282 os << "// Karatsuba recursion down to " << kLeafWidth
283 << " bit operands, below that long multiplication.\n";
284 os << "// Generated by karatsuba_gen. \n\n";
285 os << "`timescale 1ns / 1ps\n";
286 os << "`default_nettype none\n\n";
287
288 std::vector<int> widths = required_widths(n);
289 for (int w : widths) {
290 if (w <= kLeafWidth) {
291 emit_leaf(os, w);
292 } else {
293 emit_kara(os, w);
294 }
295 }
296 emit_top(os, n);
297
298 os << "\n`default_nettype wire\n";
299 return os.str();
300}
