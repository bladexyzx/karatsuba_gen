53std::string shl(const std::string& e, int from, int sh, int to) {
54 int pad = to - from - sh;
55 if (pad < 0) throw std::logic_error("shl: result does not fit");
56 if (sh == 0 && pad == 0) return e;
57 std::string s = e;
58 if (sh > 0) s = s + ", " + zeros(sh);
59 if (pad > 0) s = zeros(pad) + ", " + s;
60 return "{" + s + "}";
61}

117void emit_step(std::ostringstream& os, const Options& opt, int w, bool pipe) {
118 Split s = split_of(w);
119 int pw = 2 * w; // width of the product
120
121 // Phase 1: halves of the operands and their sums.
122 std::string a_low = "a" + rng(s.lo);
123 std::string b_low = "b" + rng(s.lo);
124 std::string a_high = "a[" + num(w - 1) + ":" + num(s.lo) + "]";
125 std::string b_high = "b[" + num(w - 1) + ":" + num(s.lo) + "]";
126 std::string a_sum = zext(a_low, s.lo, s.mid) + " + " + zext(a_high, s.hi, s.mid);
127 std::string b_sum = zext(b_low, s.lo, s.mid) + " + " + zext(b_high, s.hi, s.mid);
128 std::vector<Sig> halves = {{"a0", s.lo, a_low}, {"a1", s.hi, a_high}, {"b0", s.lo, b_low},
129 {"b1", s.hi, b_high}, {"sa", s.mid, a_sum}, {"sb", s.mid, b_sum}};
130 emit_phase(os, pipe, 1, "split, pre-additions");
131 if (pipe) {
132 emit_stage(os, halves);
133 } else {
134 emit_wires(os, halves);
135 }
136
137 // Phase 2: three half width products. In the pipelined case the
138 // sub-multipliers drive m0, m2, mm, which are registered as z0, z2, zm.
139 std::string out = "z";
140 if (pipe) out = "m";
141 emit_phase(os, pipe, 2, "three half width products");
142 os << " wire " << rng(2 * s.lo) << " " << out << "0;\n";
143 os << " wire " << rng(2 * s.hi) << " " << out << "2;\n";
144 os << " wire " << rng(2 * s.mid) << " " << out << "m;\n";
145 os << " " << mul_module(opt, s.lo) << " u_z0 (.a(a0), .b(b0), .p(" << out << "0));\n";
146 os << " " << mul_module(opt, s.hi) << " u_z2 (.a(a1), .b(b1), .p(" << out << "2));\n";
147 os << " " << mul_module(opt, s.mid) << " u_zm (.a(sa), .b(sb), .p(" << out << "m));\n\n";
148 if (pipe) {
149 emit_stage(os, {{"z0", 2 * s.lo, "m0"}, {"z2", 2 * s.hi, "m2"}, {"zm", 2 * s.mid, "mm"}});
150 }
151
152 // Phase 3: z1 = a1*b0 + a0*b1 < 2^(w+1), so the dropped high bits of zd are zero.
153 emit_phase(os, pipe, 3, "z1 = zm - z2 - z0, recombination");
154 os << " /* verilator lint_off UNUSED */\n";
155 os << " wire " << rng(2 * s.mid) << " zd = zm - " << zext("z2", 2 * s.hi, 2 * s.mid) << " - "
156 << zext("z0", 2 * s.lo, 2 * s.mid) << ";\n";
157 os << " /* verilator lint_on UNUSED */\n";
158 os << " wire " << rng(w + 1) << " z1 = zd" << rng(w + 1) << ";\n";
159 std::string product = shl("z2", 2 * s.hi, 2 * s.lo, pw) + "\n + " +
160 shl("z1", w + 1, s.lo, pw) + "\n + " +
161 zext("z0", 2 * s.lo, pw);
162 if (pipe) {
163 emit_stage(os, {{"pr", pw, product}});
164 os << " assign p = pr;\n";
165 } else {
166 os << " assign p = " << product << ";\n";
167 }
168}

191void emit_leaf(std::ostringstream& os, const Options& opt, int w) {
192 std::string comment = "// " + num(w) + " x " + num(w) + " combinational leaf multiplier\n";
193 emit_header(os, mul_module(opt, w), w, comment, "");
194 if (opt.leaf_star) {
195 os << " assign p = a * b;\n";
196 } else {
197 // Long multiplication: sum of the partial products (b[i] ? a : 0) << i.
198 os << " assign p = ";
199 for (int i = 0; i < w; ++i) {
200 if (i > 0) os << "\n + ";
201 std::string partial = "({" + num(w) + "{b[" + num(i) + "]}} & a)";
202 os << shl(partial, w, i, 2 * w);
203 }
204 os << ";\n";
205 }
206 os << "endmodule\n\n";
207}

223void emit_top(std::ostringstream& os, const Options& opt) {
224 int n = opt.n;
225 std::string comment =
226 "// " + opt.top + ": pipelined unsigned multiplier " + num(n) + " x " + num(n) + " -> " +
227 num(2 * n) + "\n" +
228 "// a, b are sampled on a rising edge of clk; p = a*b and out_valid appear\n"
229 "// after the 3rd rising edge. New operands may be applied every clock.\n"
230 "// rst_n: synchronous, active low, clears the valid chain only.\n";
231 std::string control_ports =
232 " input wire clk,\n"
233 " input wire rst_n,\n"
234 " input wire in_valid,\n"
235 " output wire out_valid,\n";
236 emit_header(os, opt.top, n, comment, control_ports);
237
238 // out_valid is in_valid delayed by three clocks
239 os << " reg [2:0] vld;\n";
240 os << " always @(posedge clk) vld <= rst_n ? {vld[1:0], in_valid} : 3'b000;\n";
241 os << " assign out_valid = vld[2];\n\n";
242
243 if (n >= 2) {
244 emit_step(os, opt, n, true);
245 } else {
246 // 1 bit: the product is a & b, delayed by three registers
247 emit_stage(os, {{"s1", 1, "a & b"}});
248 emit_stage(os, {{"s2", 1, "s1"}});
249 emit_stage(os, {{"s3", 1, "s2"}});
250 os << " assign p = {1'b0, s3};\n";
251 }
252 os << "endmodule\n";
253}

288bool valid_module_name(const std::string& s) {
289 if (s.empty()) return false;
290 if (s[0] >= '0' && s[0] <= '9') return false;
291 for (char c : s) {
292 bool letter = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z');
293 bool digit = c >= '0' && c <= '9';
294 if (!letter && !digit && c != '_') return false;
295 }
296 // words in kReserved are separated by spaces, so " name " finds whole words only
297 std::string reserved = kReserved;
298 return reserved.find(" " + s + " ") == std::string::npos;
299}

301void validate(const Options& opt) {
302 if (opt.n < 1 || opt.n > kMaxWidth) {
303 throw std::invalid_argument("N must be 1.." + num(kMaxWidth) + ", got " + num(opt.n));
304 }
305 if (opt.base < kMinBase || opt.base > kMaxBase) {
306 throw std::invalid_argument("leaf width must be " + num(kMinBase) + ".." + num(kMaxBase) +
307 ", got " + num(opt.base));
308 }
309 if (!valid_module_name(opt.top)) {
310 throw std::invalid_argument("'" + opt.top + "' is not a usable Verilog module name");
311 }

314Split split_of(int w) {
315 if (w < 2) throw std::invalid_argument("split_of: width must be at least 2");
316 Split s;
317 s.lo = (w + 1) / 2;
318 s.hi = w - s.lo;
319 s.mid = s.lo + 1;
320 return s;
321}

328std::vector<int> required_widths(const Options& opt) {
329 validate(opt);
330 std::set<int> need; // found widths, std::set keeps them sorted
331 std::vector<int> todo; // widths that still have to be split
332 if (opt.n >= 2) todo.push_back(opt.n); // the top level splits N itself
333 while (!todo.empty()) {
334 int w = todo.back();
335 todo.pop_back();
336 Split s = split_of(w);
337 std::vector<int> parts = {s.lo, s.hi, s.mid};
338 for (int part : parts) {
339 bool is_new = need.count(part) == 0;
340 need.insert(part);
341 if (is_new && part > opt.base) todo.push_back(part);
342 }
343 }
344 return std::vector<int>(need.begin(), need.end());
345}

347std::string generate_rtl(const Options& opt) {
348 validate(opt);
349 std::ostringstream os;
350 std::string leaves = "shift-add";
351 if (opt.leaf_star) leaves = "'*' operator";
352 os << "// Unsigned " << opt.n << " x " << opt.n << " -> " << 2 * opt.n
353 << " multiplier, Karatsuba algorithm, 3 stage pipeline.\n";
354 os << "// Generated by karatsuba_gen (leaf width " << opt.base << ", " << leaves
355 << " leaves). Do not edit.\n\n";
356 os << "`timescale 1ns / 1ps\n";
357 os << "`default_nettype none\n";
358 os << "/* verilator lint_off DECLFILENAME */\n\n";
359
360 std::vector<int> widths = required_widths(opt);
361 for (int w : widths) {
362 if (w <= opt.base) {
363 emit_leaf(os, opt, w);
364 } else {
365 emit_kara(os, opt, w);
366 }
367 }
368 emit_top(os, opt);
369
370 os << "\n/* verilator lint_on DECLFILENAME */\n";
371 os << "`default_nettype wire\n";
372 return os.str();
373}
