22int parse_int(const std::string& s, int lo, int hi, const std::string& what) {
23 std::string error =
24 what + " must be " + std::to_string(lo) + ".." + std::to_string(hi) + ", got '" + s + "'";
25 // at most 9 digits, so std::stoi cannot overflow
26 if (s.empty() || s.size() > 9) throw std::invalid_argument(error);
27 for (char c : s) {
28 if (c < '0' || c > '9') throw std::invalid_argument(error);
29 }
30 int v = std::stoi(s);
31 if (v < lo || v > hi) throw std::invalid_argument(error);
32 return v;
33}

35Options parse_args(const std::vector<std::string>& args) {
36 Options opt;
37 bool have_n = false;
38
39 std::size_t i = 0;
40 while (i < args.size()) {
41 std::string a = args[i];
42 bool needs_value = a == "-o" || a == "--output";
43 std::string value;
44 if (needs_value) {
45 if (i + 1 >= args.size()) throw std::invalid_argument("option " + a + " needs a value");
46 value = args[i + 1];
47 i = i + 1; // the value is consumed together with the option
48 }
49
50 if (a == "-h" || a == "--help") {
51 opt.help = true;
52 return opt;
53 } else if (a == "-o" || a == "--output") {
54 opt.out_file = value;
55 } else if (a.size() > 1 && a[0] == '-') {
56 throw std::invalid_argument("unknown option '" + a + "' (see --help)");
57 } else if (have_n) {
58 throw std::invalid_argument("unexpected argument '" + a + "'");
59 } else {
60 opt.n = parse_int(a, 1, kMaxWidth, "N");
61 have_n = true;
62 }
63 i = i + 1;
64 }
65
66 if (!have_n) throw std::invalid_argument("operand width N is missing (see --help)");
67 if (opt.out_file.empty()) opt.out_file = std::string(kTop) + ".v";
68 return opt;
69}
