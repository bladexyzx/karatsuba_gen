26int parse_int(const std::string& s, int lo, int hi, const std::string& what) {
27 std::string error =
28 what + " must be " + std::to_string(lo) + ".." + std::to_string(hi) + ", got '" + s + "'";
29 // at most 9 digits, so std::stoi cannot overflow
30 if (s.empty() || s.size() > 9) throw std::invalid_argument(error);
31 for (char c : s) {
32 if (c < '0' || c > '9') throw std::invalid_argument(error);
33 }
34 int v = std::stoi(s);
35 if (v < lo || v > hi) throw std::invalid_argument(error);
36 return v;
37}

39Options parse_args(const std::vector<std::string>& args) {
40 Options opt;
41 bool have_n = false;
42
43 std::size_t i = 0;
44 while (i < args.size()) {
45 std::string a = args[i];
46 bool needs_value = a == "-o" || a == "--output" || a == "-m" || a == "--module" ||
47 a == "-b" || a == "--base";
48 std::string value;
49 if (needs_value) {
50 if (i + 1 >= args.size()) throw std::invalid_argument("option " + a + " needs a value");
51 value = args[i + 1];
52 i = i + 1; // the value is consumed together with the option
53 }
54
55 if (a == "-h" || a == "--help") {
56 opt.help = true;
57 return opt;
58 } else if (a == "--leaf-star") {
59 opt.leaf_star = true;
60 } else if (a == "-o" || a == "--output") {
61 opt.out_file = value;
62 } else if (a == "-m" || a == "--module") {
63 if (!valid_module_name(value)) {
64 throw std::invalid_argument("'" + value + "' is not a usable Verilog module name");
65 }
66 opt.top = value;
67 } else if (a == "-b" || a == "--base") {
68 opt.base = parse_int(value, kMinBase, kMaxBase, "leaf width");
69 } else if (a.size() > 1 && a[0] == '-') {
70 throw std::invalid_argument("unknown option '" + a + "' (see --help)");
71 } else if (have_n) {
72 throw std::invalid_argument("unexpected argument '" + a + "'");
73 } else {
74 opt.n = parse_int(a, 1, kMaxWidth, "N");
75 have_n = true;
76 }
77 i = i + 1;
78 }
79
80 if (!have_n) throw std::invalid_argument("operand width N is missing (see --help)");
81 if (opt.out_file.empty()) opt.out_file = opt.top + ".v";
82 return opt;
83}
